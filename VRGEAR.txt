// VRGEAR -- the body holsters' table (first build, 2026-11-12).
//
// Read by the ENGINE (UZDXREMA src/playsim/p_vrgear.cpp). The grammar and the eight spots are fixed
// by UZDXREMA docs/plans/holsters/GEAR_CONTRACT.md, section 4. Every loaded mod may add its own
// VRGEAR lump: a later definition of the same name replaces an earlier one, `remove <Spot>` takes
// one away.
//
// Nothing here is RSQ TacticalRig's pk3: that mod is not loaded. Its NUMBERS are used, because the
// owner's TacticalRig is the floor (docs/plans/holsters/TACTICALRIG_FLOOR.md). "rig.zs" below is
// rsq_ubqseries/TacticalRig/zscript/rsqtr/rig.zs.
//
// UNITS: metres forward, left, up from the body anchor, in the torso's frame, for a 1.62 m eye
// (rig.zs:42,398). The engine scales them to the player (contract section 1).
//   at       standing position            seated   used instead while that player is seated
//   radius   the sphere a hand must be in  side     body | main | off (off: +left is toward the off hand)
//   rule     holster (a gun), summon (the grenade), zone (the pouch: no action of its own)
//   grace    tics a press still counts after the hand left (the shoulders, rig.zs:2263-2290)
//   reach    the pouch eases toward the off hand (rig.zs:342-374)
//   holds    summon: weapon class, ammo class; the first weapon class that exists wins (rig.zs:1207-1233)
//   hang     how a stored gun hangs: yaw from the body's heading, pitch nose-down, roll (rig.zs:3605-3620)
//   length   how long a stored gun is drawn, metres (0.25 default; 0.70 on the back; TacticalRig CVARINFO.txt:13,15)
//   show     a stored gun is drawn this far behind the spot and this share of its length below it (rig.zs:3625-3635)

// ---- the eight spots (positions and radii: rig.zs:86-89; seated: rig.zs:120-122) ----
spot HipL      { type gun  at  0.00  0.18 -0.70  seated 0.27  0.12 -0.67  radius 0.14  hang   0 80 -10 }
spot HipR      { type gun  at  0.00 -0.18 -0.70  seated 0.27 -0.12 -0.67  radius 0.14  hang   0 80  10 }
spot ChestL    { type gun  at  0.10  0.12 -0.46                           radius 0.11  hang   0 70  35 }
spot ChestR    { type gun  at  0.10 -0.12 -0.46                           radius 0.11  hang   0 70 -35 }
spot ShoulderL { type gun  at -0.08  0.18 -0.10                           radius 0.14  grace 10  hang -90 65 0  length 0.70  show 0.12 0.31 }
spot ShoulderR { type gun  at -0.08 -0.18 -0.10                           radius 0.14  grace 10  hang  90 65 0  length 0.70  show 0.12 0.31 }
spot Grenades  { type item at  0.12  0.00 -0.62  seated 0.13  0.00 -0.59  radius 0.08  rule summon
                 holds HandGrenades GrenadeAmmo  holds RSQGN_Grenade RSQGN_Ammo }
spot Pouch     { type item at  0.22  0.18 -0.64  seated 0.32  0.02 -0.55  radius 0.11  side off  rule zone  reach 0.035 0.06 0.005 }

// ---- what an empty hand holds ----
// Doom's Fist; Brutal Doom 22's Melee_Attacks (main hand); the BD22 VR pack's RSQ_Hands (off hand).
// A subclass counts. The rules (zscript/gear/gear.zs, EmptyMain / OffEmptyName) pick from this list:
// for the off hand, the first class here that is an off-hand weapon by default (RSQ_Hands).
emptyhand Fist Melee_Attacks RSQ_Hands

// ---- the BD22 VR pack's off-hand copies ----
// The off hand holds RSQ_Off_<gun>, a second class for the same gun. `as` files it as its gun, so a
// gun has one holster whichever hand's version is held (rig.zs:1536-1544, MainOf). The twelve pairs
// are the pack's own table (RS_Brutal22WeaponsQ 2.0.4e, src/zscript/rsq/dualtable.zs:7-16).
item RSQ_Off_BrutalPistol   { as BrutalPistol }
item RSQ_Off_Revolver       { as Revolver }
item RSQ_Off_BrutalSMG      { as BrutalSMG }
item RSQ_Off_MP40           { as MP40 }
item RSQ_Off_Rifle          { as Rifle }
item RSQ_Off_Plasma_Gun     { as Plasma_Gun }
item RSQ_Off_Shot_Gun       { as Shot_Gun }
item RSQ_Off_SSG            { as SSG }
item RSQ_Off_AssaultShotgun { as AssaultShotgun }
item RSQ_Off_RailGun        { as RailGun }
item RSQ_Off_BrutalRevolver { as BrutalRevolver }
item RSQ_Off_BrutalAxe      { as BrutalAxe }

// ---- things with a home of their own: never in a gun holster (rig.zs:823-829) ----
// TacticalRig's own list, nothing added (rig.zs:823-829). BD's HandGrenades is not in it: as in TacticalRig,
// a grenade in the hand may be stored in an empty gun holster.
ownhome rsqff_ rsqgn_ rsq_grenade rsqtr_cigweapon rsqtr_zippoweapon rsqwp_whipweapon ubqs_saber ssq_

// ---- what you feel: pulse <strength 0-1> <tics> [gap <tics>] ----
feel enter  { pulse 0.18 3 }                      // a hand enters a ring (rig.zs:866-867)
feel store  { pulse 0.55 5 }                      // a gun goes in (rig.zs:871)
feel draw   { pulse 0.65 6 }                      // a gun comes out (rig.zs:870)
feel refuse { pulse 0.30 3 gap 4 pulse 0.30 3 }   // the only double pulse (TacticalRig body.zs:782-789)

// ---- what you hear (SNDINFO.txt) ----
// store: the three holster clunks RS_Avatar already shipped; they are the same files as TacticalRig's
//        rsqtr/holster (sounds/rsqtr/rthols1-3.ogg, identical bytes).
// draw:  TacticalRig's rsqtr/draw (sounds/rsqtr/rtdraw.ogg), the owner's, copied in as RSHODRAW.
// refuse: Doom's own use-fail sound (rig.zs:877).
sound store  rs_holster_fx_store
sound draw   rs_holster_fx_draw
sound refuse *usefail
